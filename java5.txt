abstract class Activity {

    abstract void calculate();

    void display() {
        System.out.println("Carbon Footprint Tracker");
    }
}

class Car extends Activity {

    void calculate() {
        System.out.println("Car Emission: 25 kg");
    }
}

public class Main {
    public static void main(String[] args) {

        Car c = new Car();

        c.display();
        c.calculate();
    }
}
